/** §7 round 12. Pi persists images as content blocks; messageText drops them,
 *  which is why an attachment vanished on reopen.
 *  ponytail: an inline budget, because restore's whole point is a fast reopen
 *  (§17 round 10: ~25 ms). Upgrade path if this cap is ever hit in practice:
 *  emit a locator and fetch the bytes lazily instead of inlining them. */
const RESTORE_IMAGE_BUDGET = 8_000_000; // chars of base64 across one restore

export function imagesOf(content: unknown, budget: { left: number }): { urls: string[]; dropped: boolean } {
  if (!Array.isArray(content)) return { urls: [], dropped: false };
  const urls: string[] = [];
  let dropped = false;
  for (const b of content) {
    const blk = b as { type?: string; data?: string; mimeType?: string };
    if (blk.type !== "image" || !blk.data) continue;
    if (blk.data.length > budget.left) { dropped = true; continue; }
    budget.left -= blk.data.length;
    urls.push(`data:${blk.mimeType ?? "image/png"};base64,${blk.data}`);
  }
  return { urls, dropped };
}
```

/** §7 round 12: tool results carry images (7 of 8 image blocks in real session
 *  files were toolResults). They were being JSON.stringified into the details
 *  block — a megabyte of base64 where a picture belongs. */
export function resultImages(result: unknown): string[] {
  const content = (result as { content?: unknown })?.content;
  if (!Array.isArray(content)) return [];
  return content.flatMap((b) => {
    const blk = b as { type?: string; data?: string; mimeType?: string };
    return blk.type === "image" && blk.data
      ? [`data:${blk.mimeType ?? "image/png"};base64,${blk.data}`]
      : [];
  });
}

export function stripImages(result: unknown): unknown {
  const content = (result as { content?: unknown })?.content;
  if (!Array.isArray(content)) return result;
  return { ...(result as object),
           content: content.filter((b) => (b as { type?: string }).type !== "image") };
}
```

const closeChatTab = async (ws: string, paneIdx: number, tab: TabId): Promise<void> => {
  const sid = sessionOf(tab);
